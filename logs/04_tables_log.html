 221        /* Rows of n (%) for each level of a categorical variable; &ord is a numeric
 222           order variable, rows are placed after &ordbase. Zero counts are kept.       */
 223        %macro catrows(sect=, ds=, var=, ord=, ordbase=);
 224          proc sql;
 225            create table work.cf as
 226            select l.&ord as ordv, l.&var as lv, t.trtn, count(d.usubjid) as count
 227            from (select distinct &ord, &var from &ds) as l
 228                 cross join
 229                 (select distinct trtn from work.popall) as t
 230                 left join &ds as d
 231                 on d.&ord = l.&ord and d.&var = l.&var and d.trtn = t.trtn
 232            group by l.&ord, l.&var, t.trtn;
 233          quit;
 234          data work.cl(keep=roword lbl trtn cell);
 235            set work.cf;
 236            length lbl $120 cell $30;
 237            roword = &ordbase + ordv / 100;
 238            lbl    = '    ' || strip(lv);
 239            bign   = input(symget('n' || strip(put(trtn, 1.))), best12.);
 240            cell   = strip(put(count, 4.)) || ' (' || strip(put(100 * count / bign, 5.1)) || '%)';
 241          run;
 242          proc append base=work.long_&sect data=work.cl force;
 243          run;
 244        %mend catrows;
