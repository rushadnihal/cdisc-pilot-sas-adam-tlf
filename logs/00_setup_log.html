 85         /* Read each transport file and copy its dataset into the SDTM library */
 86         %macro read_xpt(name);
 87           %if %sysfunc(fileexist(&root/data/sdtm/&name..xpt)) %then %do;
 88             libname xpt xport "&root/data/sdtm/&name..xpt" access=readonly;
 89             proc copy inlib=xpt outlib=sdtm memtype=data;
 90             run;
 91             libname xpt clear;
 92           %end;
 93           %else %put ERROR: [00_setup] Missing file &root/data/sdtm/&name..xpt;
 94         %mend read_xpt;

 96         %macro read_all;
 97           %local list i n name;
 98           %let list = dm ds ex ae sv se sc qs relrec ta te ti ts tv suppdm suppae suppds;
 99           %let n = %sysfunc(countw(&list));
 100          %do i = 1 %to &n;
 101            %let name = %scan(&list, &i);
 102            %read_xpt(&name);
 103          %end;
