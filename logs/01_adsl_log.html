 224            trt01an  length=8   label='Actual Treatment for Period 01 (N)'

 234            race     length=$78 label='Race'

 236            sex      length=$1  label='Sex'
 237            ethnic   length=$25 label='Ethnicity'
 238            saffl    length=$1  label='Safety Population Flag'

 240            efffl    length=$1  label='Efficacy Population Flag'

 244            disconfl length=$1  label='Did the Subject Discontinue the Study?'

 254            durdsgr1 length=$4  label='Pooled Disease Duration Group 1'
 255            visit1dt length=8   label='Date of Visit 1' format=date9.
 256            rfstdtc  length=$10 label='Subject Reference Start Date/Time'
 257            rfendtc  length=$10 label='Subject Reference End Date/Time'
 258            visnumen length=8   label='End of Trt Visit (Vis 12 or Early Term.)'
 259            rfendt   length=8   label='Date of Discontinuation/Completion' format=date9.
 260            dcdecod  length=$27 label='Standardized Disposition Term'
 261            dcreascd length=$18 label='Reason for Discontinuation'
 262            mmsetot  length=8   label='MMSE Total'

 264        
 265          merge work.dm_base(in=in_dm)
 266                work.sv_wide
 267                work.ex_last
 268                work.ds_disp
 269                work.vs_height
 270                work.vs_weight
 271                work.sc_educ
 272                work.mh_prim
 273                work.qs_mmse
 274                work.qs_adas(in=in_adas)
 275                work.qs_cibic(in=in_cibic);
 276          by usubjid;
 277          if in_dm;
 278        
 279          /* Site group */
 280          if pool_site then sitegr1 = '900';
 281          else sitegr1 = siteid;
 282        
 283          /* Treatment: actual = planned for every subject (matches PHUSE ADSL) */
 284          trt01p = arm;
 285          select (arm);
 286            when ('Placebo')              trt01pn = 0;
 287            when ('Xanomeline Low Dose')  trt01pn = 54;
 288            when ('Xanomeline High Dose') trt01pn = 81;
 289            otherwise                     trt01pn = .;
 290          end;
 291          trt01a  = trt01p;
 292          trt01an = trt01pn;
 293        
 294          /* Dates and duration */
 295          visit1dt = v1;
 296          trtsdt   = v3;
 297          rfendt   = input(rfendtc, yymmdd10.);
 298          /* Last dose date from EX; if missing, date of discontinuation/completion */
 299          if not missing(ex_enddt) then trtedt = ex_enddt;
 300          else trtedt = rfendt;
 301          if not missing(trtsdt) and not missing(trtedt) then trtdur = trtedt - trtsdt + 1;
 302        
 303          /* Cumulative dose as planned. High dose: 54 mg/day until visit 4, 81 mg/day
 304             from visit 4 to visit 12, 54 mg/day after visit 12.                       */
 305          if not missing(trtdur) then do;
 306            if trt01pn in (0, 54) then cumdose = trt01pn * trtdur;
 307            else if trt01pn = 81 then do;
 308              if missing(v4) or trtedt < v4 then do;
 309                d1 = trtedt - trtsdt + 1;
 310                d2 = 0;
 311                d3 = 0;
 312              end;
 313              else do;
 314                d1 = v4 - trtsdt + 1;
 315                if missing(v12) or trtedt < v12 then do;
 316                  d2 = trtedt - v4;
 317                  d3 = 0;
 318                end;
 319                else do;
 320                  d2 = v12 - v4;
 321                  d3 = trtedt - v12;
 322                end;
 323              end;
 324              cumdose = 54 * d1 + 81 * d2 + 54 * d3;
 325            end;
 326            avgdd = round(cumdose / trtdur, 0.1);
 327          end;
 328        
 329          /* Age and race groupings */
 330          if not missing(age) then do;
 331            if age < 65 then agegr1n = 1;
 332            else if age <= 80 then agegr1n = 2;
 333            else agegr1n = 3;
 334          end;
 335          select (agegr1n);
 336            when (1) agegr1 = '<65';
 337            when (2) agegr1 = '65-80';
 338            when (3) agegr1 = '>80';
 339            otherwise agegr1 = ' ';
 340          end;
 341          select (race);
 342            when ('WHITE')                             racen = 1;
 343            when ('BLACK OR AFRICAN AMERICAN')         racen = 2;
 344            when ('ASIAN')                             racen = 3;
 345            when ('AMERICAN INDIAN OR ALASKA NATIVE')  racen = 6;
 346            otherwise                                  racen = .;
 347          end;
 348        
 349          /* Population flags */
 350          if armcd ne ' ' then ittfl = 'Y'; else ittfl = 'N';
 351          if ittfl = 'Y' and not missing(trtsdt) then saffl = 'Y'; else saffl = 'N';
 352          if saffl = 'Y' and in_adas and in_cibic then efffl = 'Y'; else efffl = 'N';
 353          if not missing(v8)  and rfendt >= v8  then comp8fl  = 'Y'; else comp8fl  = 'N';
 354          if not missing(v10) and rfendt >= v10 then comp16fl = 'Y'; else comp16fl = 'N';
 355          if not missing(v12) and rfendt >= v12 then comp24fl = 'Y'; else comp24fl = 'N';
 356        
 357          /* Disposition */
 358          select (dsdecod);
 359            when ('ADVERSE EVENT')               dcreascd = 'Adverse Event';
 360            when ('COMPLETED')                   dcreascd = 'Completed';
 361            when ('DEATH')                       dcreascd = 'Death';
 362            when ('LACK OF EFFICACY')            dcreascd = 'Lack of Efficacy';
 363            when ('LOST TO FOLLOW-UP')           dcreascd = 'Lost to Follow-up';
 364            when ('PHYSICIAN DECISION')          dcreascd = 'Physician Decision';
 365            when ('STUDY TERMINATED BY SPONSOR') dcreascd = 'Sponsor Decision';
 366            when ('WITHDRAWAL BY SUBJECT')       dcreascd = 'Withdrew Consent';
 367            when ('PROTOCOL VIOLATION') do;
 368              if dsterm = 'PROTOCOL ENTRY CRITERIA NOT MET' then dcreascd = 'I/E Not Met';
 369              else dcreascd = 'Protocol Violation';
 370            end;
 371            otherwise dcreascd = ' ';
 372          end;
 373          dcdecod = dsdecod;
