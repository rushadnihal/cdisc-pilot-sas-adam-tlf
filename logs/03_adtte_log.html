 109        /* ---- 3. Derive ADTTE ------------------------------------------------------------ */
 110        data work.adtte;
 111          length param $40 paramcd $8 evntdesc $40 srcdom $4 srcvar $6;
 112        
 113          merge work.adsl_sub(in=in_adsl)
 114                work.first_derm(in=in_event);
 115          by usubjid;
 116          if in_adsl;
 117        
 118          param    = 'Time to First Dermatologic Event';
 119          paramcd  = 'TTDE';
 120          startdt  = trtsdt;
 121        
 122          if in_event then do;
 123            adt      = ev_dt;
 124            cnsr     = 0;
 125            evntdesc = 'Dermatologic Event Occurred';
 126            srcdom   = 'ADAE';
 127            srcvar   = 'ASTDT';
 128            srcseq   = ev_seq;
 129          end;
 130          else do;
 131            adt      = rfendt;
 132            cnsr     = 1;
 133            evntdesc = 'Study Completion Date';
 134            srcdom   = 'ADSL';
 135            srcvar   = 'RFENDT';
 136          end;
 137        
 138          if not missing(adt) and not missing(startdt) then aval = adt - startdt + 1;
 139        
 140          format startdt adt date9.;
 141          label param    = 'Parameter Description'
 142                paramcd  = 'Parameter Code'
 143                aval     = 'Analysis Value'
 144                startdt  = 'Time to Event Origin Date for Subject'
 145                adt      = 'Analysis Date'
 146                cnsr     = 'Censor'
 147                evntdesc = 'Event or Censoring Description'
 148                srcdom   = 'Source Domain'
 149                srcvar   = 'Source Variable'
 150                srcseq   = 'Source Sequence Number'
 151                trtp     = 'Planned Treatment'
 152                trta     = 'Actual Treatment'
 153                trtan    = 'Actual Treatment (N)';
 154        run;
